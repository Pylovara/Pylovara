// C:   int main() { ... return 0; }

/* ===== PROTEINE (AUSFÜHRBAR) ===== */
// MCS: »—[ ... ]«
// C:   ausfuehrbarer_code_block();

/* ===== PROTEINE (NICHT AUSFÜHRBAR) ===== */
// MCS: —[ ... ]
// C:   // Kommentar oder Datenstruktur

/* ===== PROTEINE (OFF-PIPE) ===== */
// MCS: »[ ... ]«
// C:   ausfuehrbarer_code_block_ohne_anschluss();

/* ===== ZEILENUMBRUCH ===== */
// MCS: ¬·
// C:   ;

/* ===== SENTIATOR WENN-KANN ===== */
// MCS: ¶
// C:   if (bedingung == true) {

/* ===== SENTIATOR WENN-NICHT ===== */
// MCS: ¶¶
// C:   } else {

/* ===== NITROX EXTRAKTOR (FF) ===== */
// MCS: ± [(<NR>)—Ø ¥ <CACHE> ¥ Ø]
// C:   cache_speichern(<NR>, <CACHE>);

/* ===== NITROX INJEKTOR (FT) ===== */
// MCS: ÷ [Ø ¥ <CACHE> ¥ Ø—(<NR>)]
// C:   cache_laden(<NR>, &<CACHE>);

/* ===== SHELL-BOXI ===== */
// MCS: »[' <COMMAND> ']«
// C:   system("<COMMAND>");

/* ===== PFAD-BOXI ===== */
// MCS: »[$ <PFAD> $]«
// C:   FILE* f = fopen("<PFAD>", "r");

/* ===== NET-BOXI ===== */
// MCS: »[§ <IP> §]«
// C:   socket_connect("<IP>");

/* ===== ALU-BOXI ===== */
// MCS: »[× <AUSDRUCK> ×]«
// C:   int ergebnis = <AUSDRUCK>;

/* ===== MCD-BOXI ===== */
// MCS: »[~ <BEFEHL> ~]«
// C:   mcd_befehl("<BEFEHL>");

/* ===== SYNAPTOR ===== */
// MCS: ¿¹
// C:   sync_wait();

/* ===== ZYTRON OUTPUT ===== */
// MCS: £›
// C:   zytron_ausgabe();

/* ===== ZYTRON INPUT ===== */
// MCS: £‹
// C:   zytron_eingabe();

/* ===== ARGUMENTE ===== */
// MCS: ««V<NR>
// C:   variable_<NR>;

/* ===== KETTE ===== */
// MCS: ⌬
// C:   // verkettung
